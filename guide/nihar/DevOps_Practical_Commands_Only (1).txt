
==================================================
1. DOCKER: BUILD AND RUN A MAVEN WEB APPLICATION
==================================================

cd maven_web
mvn clean package
ls target

nano Dockerfile
Dockerfile:

FROM tomcat:10.1-jdk21-temurin
COPY target/*.war /usr/local/tomcat/webapps/ROOT.war
EXPOSE 8080
CMD ["catalina.sh", "run"]

docker build -t maven-app .
docker images

docker rm -f maven-app
docker run -d -p 8082:8080 --name maven-app maven-app
docker ps

http://localhost:8082

STEP 11: Stop and remove the container

docker stop maven-app
docker rm maven-app
